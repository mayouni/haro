~~~ The fixtures of rung 18 (Haro v0.18): the structure level of the computing altitude, for the Ring++ face.
~~~
~~~ Hand-composed BEFORE the abstraction or the printer existed (the W discipline: fixtures first, code second). Each one
~~~ is a Ring program, the L1 graph it must abstract to, and the canonical text the printer must print. Where a fixture
~~~ turns on how the PARSER reads a shape (a comment between statements, a named parameter, a bare return), the shape was
~~~ read off `rnxc dump` first, never off the implementation.
~~~
~~~ Format: `=== NN name`, then `~~~ ` notes (`trap:` names the Ring finding; `dis: yes` also compiles original and canonical
~~~ and compares `rnxc dis`), then `--- ring` (the program; absent for a printer-only fixture), `--- l1` (the graph, as the
~~~ s-expressions of v18/GRAMMAR.md), `--- out` (the canonical text).
~~~

=== 02 spellings
~~~ the grammar's alternative spellings print as one: FUNCTION/FUNC/DEF, IF..ENDIF/OK, ELSEIF/BUT, OTHER/ELSE, GET/GIVE,
~~~ BREAK/EXIT, CONTINUE/LOOP, FOREACH/FOR, ENDFOR/NEXT, ENDWHILE, ENDSWITCH/OFF, ENDTRY/DONE, ENDFUNC, NOT; keywords in capitals.
~~~ (Changed after the court ran: PUT and SEE are left out, because Ring prints no line for them and rnxc does -- fixture 22.)
~~~ dis: yes
--- ring
FUNCTION Main
	IF a = 1 AND b != 2 OR NOT c
		? "x"
	ELSEIF d
		GET e
	OTHER
		BREAK
	ENDIF
	FOREACH i IN list
		CONTINUE
	ENDFOR
	WHILE x ENDWHILE
	SWITCH y CASE 1 ? 1 OTHER ? 2 ENDSWITCH
	TRY f() CATCH g() ENDTRY
ENDFUNC
--- l1
(file :items [
  (func :name "Main" :items [
    (if :cond (bin :op "or"
                   :l (bin :op "and" :l (bin :op "=" :l (id "a") :r (num "1")) :r (bin :op "!=" :l (id "b") :r (num "2")))
                   :r (un :op "not" :e (id "c")))
        :then [(say :e (str "\"x\""))]
        :elifs [(elif :cond (id "d") :items [(give :target (id "e"))])]
        :else [(exit)])
    (for :var "i" :kind "in" :a (id "list") :items [(loop)])
    (while :cond (id "x"))
    (switch :subject (id "y")
            :cases [(case :values [(num "1")] :items [(say :e (num "1"))])]
            :other [(say :e (num "2"))])
    (try :items [(expr :e (call :callee (id "f")))]
         :catch (catch :items [(expr :e (call :callee (id "g")))]))])])
--- out
func Main()
	if a = 1 and b != 2 or not c
		? "x"
	but d
		give e
	else
		exit
	ok
	for i in list
		loop
	next
	while x
	end
	switch y
		on 1
			? 1
		other
			? 2
	off
	try
		f()
	catch
		g()
	done

=== 06 statements
~~~ trap: F-74 -- `exit 2` leaves two loops; the level is an expression and is kept
~~~ dis: yes
--- ring
func Main
	if a = 1 ? "one" but a = 2 ? "two" else ? "many" ok
	while i < 3 i++ end
	for i = 1 to 10 step 2 loop next
	for k in items exit 2 next
	do i-- again i > 0
	switch a on 1, 2 ? "ab" other ? "c" off
	try risky() catch(e) ? e finally cleanup() done
--- l1
(file :items [
  (func :name "Main" :items [
    (if :cond (bin :op "=" :l (id "a") :r (num "1"))
        :then [(say :e (str "\"one\""))]
        :elifs [(elif :cond (bin :op "=" :l (id "a") :r (num "2")) :items [(say :e (str "\"two\""))])]
        :else [(say :e (str "\"many\""))])
    (while :cond (bin :op "<" :l (id "i") :r (num "3")) :items [(expr :e (post :op "++" :e (id "i")))])
    (for :var "i" :kind "to" :a (num "1") :b (num "10") :step (num "2") :items [(loop)])
    (for :var "k" :kind "in" :a (id "items") :items [(exit :e (num "2"))])
    (dowhile :items [(expr :e (post :op "--" :e (id "i")))] :cond (bin :op ">" :l (id "i") :r (num "0")))
    (switch :subject (id "a")
            :cases [(case :values [(num "1") (num "2")] :items [(say :e (str "\"ab\""))])]
            :other [(say :e (str "\"c\""))])
    (try :items [(expr :e (call :callee (id "risky")))]
         :catch (catch :binding "e" :items [(say :e (id "e"))])
         :finally [(expr :e (call :callee (id "cleanup")))])])])
--- out
func Main()
	if a = 1
		? "one"
	but a = 2
		? "two"
	else
		? "many"
	ok
	while i < 3
		i++
	end
	for i = 1 to 10 step 2
		loop
	next
	for k in items
		exit 2
	next
	do
		i--
	again i > 0
	switch a
		on 1, 2
			? "ab"
		other
			? "c"
	off
	try
		risky()
	catch(e)
		? e
	finally
		cleanup()
	done

=== 07 expressions
~~~ the structure of an expression is kept as written, parentheses included: the graph does not re-associate
~~~ the right side of an assignment cannot assign: in `q = a = b = c` the first `=` assigns and the others compare, left to right
~~~ dis: yes
--- ring
x = a + b * c
y = (a + b) * -c
z = not a and b or c
w = a.b(c)[d].e
v = obj.method(1, "s")
n += 1
r = [1:5]
i++
q = a = b = c
--- l1
(file :items [
  (expr :e (assign :op "=" :l (id "x") :r (bin :op "+" :l (id "a") :r (bin :op "*" :l (id "b") :r (id "c")))))
  (expr :e (assign :op "=" :l (id "y") :r (bin :op "*" :l (paren :e (bin :op "+" :l (id "a") :r (id "b"))) :r (un :op "-" :e (id "c")))))
  (expr :e (assign :op "=" :l (id "z") :r (bin :op "or" :l (bin :op "and" :l (un :op "not" :e (id "a")) :r (id "b")) :r (id "c"))))
  (expr :e (assign :op "=" :l (id "w")
                   :r (member :obj (index :obj (call :callee (member :obj (id "a") :prop (id "b")) :args [(id "c")]) :idx (id "d")) :prop (id "e"))))
  (expr :e (assign :op "=" :l (id "v") :r (call :callee (member :obj (id "obj") :prop (id "method")) :args [(num "1") (str "\"s\"")])))
  (expr :e (assign :op "+=" :l (id "n") :r (num "1")))
  (expr :e (assign :op "=" :l (id "r") :r (list :items [(bin :op ":" :l (num "1") :r (num "5"))])))
  (expr :e (post :op "++" :e (id "i")))
  (expr :e (assign :op "=" :l (id "q") :r (bin :op "=" :l (bin :op "=" :l (id "a") :r (id "b")) :r (id "c"))))])
--- out
x = a + b * c
y = (a + b) * -c
z = not a and b or c
w = a.b(c)[d].e
v = obj.method(1, "s")
n += 1
r = [1:5]
i++
q = a = b = c

=== 11 comments
~~~ a comment is a node, never layout: kept as an item where it stands, as the trailing comment of the statement on its line, and -- when it
~~~ sat inside an expression -- as an item after that statement (changed after the court ran: a layout-free text has nowhere to keep it
~~~ inside, so the first version of this fixture, with the comment as a field of the statement, could not survive the round trip). A
~~~ comment after the last statement of a function that no `end` closes is the parser's, and belongs to the file: so is `# inner` here,
~~~ the last thing the function holds once it is hoisted (changed after the court ran, again: the first expectation kept it inside)
~~~ dis: yes
--- ring
# header
func Main
	x = 1  # trailing
	# own line
	y = 2
	#--> a promise
	if y = 2
		# dangling in if
	ok
	/* block
	   comment */
	z = f(1, # inner
	      2)
	# dangling at end
--- l1
(file :items [
  (comment "# header")
  (func :name "Main" :items [
    (expr :e (assign :op "=" :l (id "x") :r (num "1")) :trail (comment "# trailing"))
    (comment "# own line")
    (expr :e (assign :op "=" :l (id "y") :r (num "2")))
    (comment "#--> a promise")
    (if :cond (bin :op "=" :l (id "y") :r (num "2")) :then [(comment "# dangling in if")])
    (comment "/* block\n\t   comment */")
    (expr :e (assign :op "=" :l (id "z") :r (call :callee (id "f") :args [(num "1") (num "2")])))])
  (comment "# inner")
  (comment "# dangling at end")])
--- out
# header
func Main()
	x = 1  # trailing
	# own line
	y = 2
	#--> a promise
	if y = 2
		# dangling in if
	ok
	/* block
	   comment */
	z = f(1, 2)

# inner
# dangling at end

=== 12 a bare return
~~~ a `return` that ends its line is BARE: Ring's parser looks at the very next token, finds the end of the line, and returns NULL
~~~ (vendor/ring-vm/src/stmt.c:931), so the expression on the next row is a statement of its own that never runs. Measured on the oracle
~~~ and the official 1.27: F() answers the empty string. The grammar hangs that expression on the `return` anyway, and rnxc did the same
~~~ until 2026-10-04, and the first version of this fixture said the compiler joins them, judged by `rnxc dis` alone. (Changed after the
~~~ court ran, twice: from the parser's reading to rnxc's, and from rnxc's to Ring's.) The graph keeps the dead statement, because the text has it.
~~~ dis: yes
~~~ run: yes
--- ring
? "[" + F() + "]"
func F
	return
	x = 1
--- l1
(file :items [
  (say :e (bin :op "+" :l (bin :op "+" :l (str "\"[\"") :r (call :callee (id "F"))) :r (str "\"]\"")))
  (func :name "F" :items [(return) (expr :e (assign :op "=" :l (id "x") :r (num "1")))])])
--- out
? "[" + F() + "]"

func F()
	return
	x = 1

=== 17 braces, as another spelling of a block
~~~ the `{ }` bodies of a class, function, if and while are spellings of the keyword blocks. The braces close nothing as far as Ring's flow
~~~ goes (measured: v18/probes/fall.py): the statements after the class run as part of the last method's flow, so the graph says so
~~~ (changed after the court ran: the first version closed the class with `end`, which Ring ignores). Judged by running both texts on the
~~~ oracle, not by `rnxc dis`: rnxc types a method whose `return 1` the parser split from its value as `ret=str` and the braced one as
~~~ `ret=ref`, though both answer 1 (probed) -- a finding about rnxc, not about the graph
~~~ run: yes
--- ring
o = new A
? o.F()
class A {
	x = 1
	func F() { return 1 }
}
if a { b = 1 }
while c { d = 2 }
--- l1
(file :items [
  (expr :e (assign :op "=" :l (id "o") :r (new :name ["A"])))
  (say :e (call :callee (member :obj (id "o") :prop (id "F"))))
  (class :name "A" :items [
    (expr :e (assign :op "=" :l (id "x") :r (num "1")))
    (func :name "F" :items [
      (return :e (num "1"))
      (if :cond (id "a") :then [(expr :e (assign :op "=" :l (id "b") :r (num "1")))])
      (while :cond (id "c") :items [(expr :e (assign :op "=" :l (id "d") :r (num "2")))])])])])
--- out
o = new A
? o.F()

class A
	x = 1

	func F()
		return 1
		if a
			b = 1
		ok
		while c
			d = 2
		end

=== 19 a printer-only fixture: parentheses by precedence
~~~ a graph a refinement builds has no parenthesized nodes; the printer adds the parentheses the structure needs, and no others. Left
~~~ association: `a - (b - c)` needs them, `(a - b) - c` does not. The statements are assignments because a bare expression that begins
~~~ with `-` joins the line above as a subtraction -- Ring reads across lines, and the first version of this fixture, six bare
~~~ expressions, said so (changed after the court ran; the hazard is in GRAMMAR.md's collision table)
--- l1
(file :items [
  (expr :e (assign :op "=" :l (id "r") :r (bin :op "*" :l (bin :op "+" :l (id "a") :r (id "b")) :r (id "c"))))
  (expr :e (assign :op "=" :l (id "r") :r (bin :op "-" :l (id "a") :r (bin :op "-" :l (id "b") :r (id "c")))))
  (expr :e (assign :op "=" :l (id "r") :r (bin :op "-" :l (bin :op "-" :l (id "a") :r (id "b")) :r (id "c"))))
  (expr :e (assign :op "=" :l (id "r") :r (un :op "-" :e (bin :op "+" :l (id "a") :r (id "b")))))
  (expr :e (assign :op "=" :l (id "r") :r (bin :op "and" :l (bin :op "or" :l (id "a") :r (id "b")) :r (id "c"))))
  (expr :e (assign :op "=" :l (id "r") :r (bin :op "+" :l (id "a") :r (bin :op "*" :l (id "b") :r (id "c")))))])
--- out
r = (a + b) * c
r = a - (b - c)
r = a - b - c
r = -(a + b)
r = (a or b) and c
r = a + b * c

~~~ statement after a definition, and the printer never writes an `end`: this fixture's first version had the printer closing a

end

	y = 3
	func G()
		return 4
	end
	z = 5

(file :items [
  (func :name "F" :items [
    (return :e (num "1"))
    (expr :e (assign :op "=" :l (id "x") :r (num "2")))])
  (class :name "A" :items [
    (expr :e (assign :op "=" :l (id "y") :r (num "3")))
    (func :name "G" :items [
      (return :e (num "4"))
      (expr :e (assign :op "=" :l (id "z") :r (num "5")))])])])
--- out
func F()
	return 1
	x = 2

class A
	y = 3

	func G()
		return 4

=== 21 operators that keep their spelling
~~~ `!` and `not` compile alike and print as `not`; `&&` and `||` do NOT compile like `and` and `or` (rnxc traps on them), so they keep
~~~ their spelling: a spelling is normalized only where the compiler has been measured to read the two alike
~~~ dis: yes
--- ring
x = a && b
y = a || b
z = !c
w = a AND b OR NOT c

(file :items [
  (expr :e (assign :op "=" :l (id "x") :r (bin :op "&&" :l (id "a") :r (id "b"))))
  (expr :e (assign :op "=" :l (id "y") :r (bin :op "||" :l (id "a") :r (id "b"))))
  (expr :e (assign :op "=" :l (id "z") :r (un :op "not" :e (id "c"))))
  (expr :e (assign :op "=" :l (id "w") :r (bin :op "or" :l (bin :op "and" :l (id "a") :r (id "b")) :r (un :op "not" :e (id "c")))))])

=== 23 a unary that begins a statement binds to its first operand
~~~ the grammar reads a statement that BEGINS with a unary `+` or `-` as that operator over the whole rest when the rest holds a binary `+` or
~~~ `-` (`-o - 3` is `-(o - 3)`), and Ring reads it as `(-o) - 3`: unary binds tighter than any binary operator. Measured on the oracle and
~~~ the official 1.27, which agree, with an object whose `operator` prints its name: `-o - 3` calls neg and then `-`, `-o + 1 - 2` calls neg,
~~~ `+` and `-`. rnxc follows the grammar and calls them in the other order, and so did the first printer of this rung, which wrote
~~~ `-(o - 3)`: every judge that reads the tree (`rnxc dis`, the graph read back) agreed with it and Ring did not. The statement's value is
~~~ discarded, so only an overloaded operator can tell -- which is why the court RUNS both texts on the oracle. (Added after the court's first
~~~ run on `rnxc fmt`, HF-79; the first version of this fixture's graph was the grammar's, and the court said so.)
~~~ run: yes
--- ring
o = new K
-o - 3
-o + 1 - 2
class K
	func operator cOp, v
		? "op " + cOp
		return This
--- l1
(file :items [
  (expr :e (assign :op "=" :l (id "o") :r (new :name ["K"])))
  (expr :e (bin :op "-" :l (un :op "-" :e (id "o")) :r (num "3")))
  (expr :e (bin :op "-" :l (bin :op "+" :l (un :op "-" :e (id "o")) :r (num "1")) :r (num "2")))
  (class :name "K" :items [
    (func :name "operator" :params [(param :name "cOp") (param :name "v")] :items [
      (say :e (bin :op "+" :l (str "\"op \"") :r (id "cOp")))
      (return :e (self))])])])
--- out
o = new K
-o - 3
-o + 1 - 2

class K
	func operator(cOp, v)
		? "op " + cOp
		return This
